# Utilities are run in [Windows cmd]

# Connect to DB using psql
"D:\Program Files\PostgreSQL\17\bin\psql.exe" -h [hostname] -p [port] -U [username] -d [database_name]

# Connect to DB using psql [Multi-line example]
"D:\Program Files\PostgreSQL\17\bin\psql.exe" ^
-h [hostname] ^
-p [port] ^
-U [username] ^
-d [database_name]


# Connect to DB using psql and execute query command
# --pset=pager=off, turns off paginated results to show all results
"D:\Program Files\PostgreSQL\17\bin\psql.exe" -h [hostname] -p [port] -U [username] -d [database_name] --pset=pager=off -c "[query]"

# Connect to DB using psql and execute query command [Multi-line example]
# --pset=pager=off, turns off paginated results to show all results
"D:\Program Files\PostgreSQL\17\bin\psql.exe" ^
-h [hostname] ^
-p [port] ^
-U [username] ^
-d [database_name] ^
--pset=pager=off ^
-c "[query]"


# Connect to DB using psql and execute query and output lines of query result that contain specified string 
# --pset=pager=off, turns off paginated results to show all results
# pipe findstr "[string]" will filter query results based on specified string
"D:\Program Files\PostgreSQL\17\bin\psql.exe" -h [hostname] -p [port] -U [username] -d [database_name] --pset=pager=off -c "[query]" | findstr "cost="

# Connect to DB using psql and execute query and output lines of query result that contain specified string [Multi-line example]
# --pset=pager=off, turns off paginated results to show all results
# pipe findstr "[string]" will filter query results based on specified string
"D:\Program Files\PostgreSQL\17\bin\psql.exe" ^
-h [hostname] ^
-p [port] ^
-U [username] ^
-d [database_name] ^
--pset=pager=off ^
-c "[query]" | findstr "cost="


# ANALYZE tables in bulk using vacuumdb utility
"D:\Program Files\PostgreSQL\17\bin\vacuumdb.exe" -h [hostname] -p [port] -U [username] -d [database_name] --analyze-only -j 8


# vacuumlo (vacuum utility for orphaned large objects)
# -n means dry run
"D:\Program Files\PostgreSQL\17\bin\vacuumlo.exe" -n -h [host] -p [port] -U [user] [database_name]

# vacuumlo actual run, -n flag removed
"D:\Program Files\PostgreSQL\17\bin\vacuumlo.exe" -h [host] -p [port] -U [user] [database_name]


# BACKUP using pg_dump

# pg_dump for backing up database (schema-only) [Directory Dump]
"D:\Program Files\PostgreSQL\17\bin\pg_dump.exe" -h [hostname] -p [port] -U [username] -d [database_name] --schema-only -j 4 --verbose -F d -f  "path\database_backup_schema"

# pg_dump for backing up database (data-only) [Directory Dump]
"D:\Program Files\PostgreSQL\17\bin\pg_dump.exe" -h [hostname] -p [port] -U [username] -d [database_name] --data-only -j 4 --verbose -F d -f  "path\database_backup_data"

# pg_dump for backing up schema (schema-only) [Directory Dump]
"D:\Program Files\PostgreSQL\17\bin\pg_dump.exe" -h [hostname] -p [port] -U [username] -d [database_name] -n [schema_name] --schema-only -j 4 --verbose -F d -f  "path\schema_backup_schema"

# pg_dump for backing up schema (data-only) [Directory Dump]
"D:\Program Files\PostgreSQL\17\bin\pg_dump.exe" -h [hostname] -p [port] -U [username] -d [database_name] -n [schema_name] --data-only -j 4 --verbose -F d -f  "path\schema_backup_data"

# pg_dump for backing up partitioned table (table schema-only) [Directory Dump]
"D:\Program Files\PostgreSQL\17\bin\pg_dump.exe" -h [hostname] -p [port] -U [username] -d [database_name] --table-and-children [schema.parent_table_name] --schema-only -j 4 --verbose -F d -f  "path\parent_table_backup_schema"

# pg_dump for backing up partitioned table (table data-only) [Directory Dump]
"D:\Program Files\PostgreSQL\17\bin\pg_dump.exe" -h [hostname] -p [port] -U [username] -d [database_name] --table-and-children [schema.parent_table_name] --data-only -j 4 --verbose -F d -f  "path\parent_table_backup_data"
